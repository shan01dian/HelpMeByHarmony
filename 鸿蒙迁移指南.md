```typescript
import { http } from '@kit.NetworkKit';
import { preferences } from '@kit.ArkData';

const API_BASE = 'https://your-api-domain.com'; // 生产环境地址

// 全局 HTTP 单例
let httpRequest: http.HttpRequest = http.createHttp();

/**
 * 封装 GET 请求
 */
export async function httpGet<T>(path: string, params?: Record<string, string>): Promise<T> {
  let url = `${API_BASE}${path}`;
  if (params) {
    const query = Object.entries(params)
      .map(([k, v]) => `${encodeURIComponent(k)}=${encodeURIComponent(v)}`)
      .join('&');
    url += `?${query}`;
  }

  // 获取 token
  const token = await getToken();
  const header: Record<string, string> = {
    'Content-Type': 'application/json'
  };
  if (token) {
    header['Authorization'] = `Bearer ${token}`;
  }

  const response = await httpRequest.request(url, {
    method: http.RequestMethod.GET,
    header: header,
    readTimeout: 15000,
    connectTimeout: 15000
  });

  if (response.responseCode === 200) {
    return JSON.parse(response.result as string) as T;
  } else if (response.responseCode === 401) {
    // token 过期，跳转登录
    await clearToken();
    throw new Error('UNAUTHORIZED');
  } else {
    throw new Error(`HTTP ${response.responseCode}: ${response.result}`);
  }
}
```

**Step 2：封装图片上传**

```typescript
// entry/src/main/ets/services/UploadService.ets
import { http } from '@kit.NetworkKit';
import { photoAccessHelper } from '@kit.CoreFileKit';
import { fileIo } from '@kit.CoreFileKit';
import { getToken } from './HttpUtil';

const API_BASE = 'https://your-api-domain.com';

/**
 * 上传多张图片，返回 URL 数组
 */
export async function uploadImages(fileUris: string[]): Promise<string[]> {
  const token = await getToken();
  const httpRequest = http.createHttp();

  const results: string[] = [];

  for (const uri of fileUris) {
    // 读取文件
    const file = fileIo.openSync(uri, fileIo.OpenMode.READ_ONLY);
    const stat = fileIo.statSync(file.fd);
    const buffer = new ArrayBuffer(stat.size);
    fileIo.readSync(file.fd, buffer);
    fileIo.closeSync(file);

    // 构建 multipart/form-data
    const boundary = '----HelpMeBoundary' + Date.now();
    const fileName = `img_${Date.now()}.jpg`;

    const bodyParts: ArrayBuffer[] = [];

    // 文件部分
    const fileHeader = `--${boundary}\r\nContent-Disposition: form-data; name="images"; filename="${fileName}"\r\nContent-Type: image/jpeg\r\n\r\n`;
    bodyParts.push(stringToArrayBuffer(fileHeader));
    bodyParts.push(buffer);
    bodyParts.push(stringToArrayBuffer('\r\n'));

    const endBoundary = `--${boundary}--\r\n`;
    bodyParts.push(stringToArrayBuffer(endBoundary));

    // 合并 buffer
    const totalLength = bodyParts.reduce((sum, part) => sum + part.byteLength, 0);
    const mergedBuffer = new ArrayBuffer(totalLength);
    const uint8View = new Uint8Array(mergedBuffer);
    let offset = 0;
    for (const part of bodyParts) {
      uint8View.set(new Uint8Array(part), offset);
      offset += part.byteLength;
    }

    const response = await httpRequest.request(`${API_BASE}/upload/images`, {
      method: http.RequestMethod.POST,
      header: {
        'Content-Type': `multipart/form-data; boundary=${boundary}`,
        'Authorization': `Bearer ${token}`
      },
      extraData: mergedBuffer,
      readTimeout: 30000,
      connectTimeout: 15000
    });

    if (response.responseCode === 200) {
      const result = JSON.parse(response.result as string);
      results.push(...result.urls);
    } else {
      throw new Error('图片上传失败');
    }
  }

  httpRequest.destroy();
  return results;
}

function stringToArrayBuffer(str: string): ArrayBuffer {
  const encoder = new util.TextEncoder();
  return encoder.encodeInto(str).buffer;
}

/**
 * 从系统相册选择图片
 */
export async function selectImages(maxCount: number = 9): Promise<string[]> {
  const photoHelper = photoAccessHelper.getPhotoAccessHelper(getContext());

  const result = await photoAccessHelper.PhotoSelectOptions.newBuilder()
    .setMIMEType('image/*')
    .setMaxSelectNumber(maxCount)
    .build();

  const selectResult = await photoHelper.select(result);
  return selectResult.photoUris;
}
```

### 4.3 认证模块迁移

#### 原始代码分析

原项目使用手机验证码登录，流程为：
1. 输入手机号 → 图形验证码 → 发送短信
2. 输入短信验证码 → 登录/注册
3. JWT token 存储在 localStorage

#### 4.6.1 WebSocket 服务封装

```typescript
// entry/src/main/ets/services/WebSocketService.ets
import { webSocket } from '@kit.NetworkKit';
import { getToken } from './HttpUtil';

const WS_BASE = 'wss://your-api-domain.com';

export interface ChatMessage {
  roomId: string;
  senderId: number;
  messageType: 'text' | 'image' | 'location';
  text?: string;
  imageUrl?: string;
  location?: { lng: number; lat: number; address: string };
  sendTime: string;
  userName: string;
  targetUserId?: number;
}

type MessageHandler = (msg: ChatMessage) => void;
type StatusHandler = (status: string) => void;

class WebSocketService {
  private ws: webSocket.WebSocket | null = null;
  private messageHandlers: Map<string, MessageHandler[]> = new Map();
  private statusHandlers: StatusHandler[] = [];
  private reconnectAttempts: number = 0;
  private maxReconnect: number = 5;
  private currentRoomId: string = '';

  /**
   * 连接 WebSocket
   */
  async connect(): Promise<void> {
    const token = await getToken();
    if (!token) return;

    this.ws = webSocket.createWebSocket();

    this.ws.on('open', () => {
      console.info('WebSocket 已连接');
      this.reconnectAttempts = 0;
      this.notifyStatus('connected');

      // 连接成功后认证（模拟 Socket.IO 的 auth 机制）
      this.send('authenticate', { token });
    });

    this.ws.on('message', (data: string | ArrayBuffer) => {
      if (typeof data === 'string') {
        try {
          const parsed = JSON.parse(data);
          this.handleMessage(parsed.event, parsed.data);
        } catch {
          console.error('WebSocket 消息解析失败');
        }
      }
    });

    this.ws.on('close', () => {
      console.info('WebSocket 已断开');
      this.notifyStatus('disconnected');
      this.tryReconnect();
    });

    this.ws.on('error', (err: Error) => {
      console.error('WebSocket 错误:', err.message);
      this.notifyStatus('error');
    });

    // 连接（携带 token）
    await this.ws.connect(`${WS_BASE}?token=${token}`);
  }

  /**
   * 加入聊天房间
   */
  joinRoom(roomId: string): void {
    this.currentRoomId = roomId;
    this.send('joinRoom', { roomId });
  }

  /**
   * 发送聊天消息
   */
  sendChatMessage(msg: {
    roomId: string;
    messageType: string;
    text?: string;
    imageUrl?: string;
    location?: object;
  }): void {
    this.send('chat message', msg);
  }

  /**
   * 监听消息
   */
  on(event: string, handler: MessageHandler): void {
    if (!this.messageHandlers.has(event)) {
      this.messageHandlers.set(event, []);
    }
    this.messageHandlers.get(event)!.push(handler);
  }

  /**
   * 监听连接状态
   */
  onStatusChange(handler: StatusHandler): void {
    this.statusHandlers.push(handler);
  }

  /**
   * 移除监听
   */
  off(event: string, handler: MessageHandler): void {
    const handlers = this.messageHandlers.get(event);
    if (handlers) {
      const idx = handlers.indexOf(handler);
      if (idx >= 0) handlers.splice(idx, 1);
    }
  }

  /**
   * 断开连接
   */
  disconnect(): void {
    if (this.ws) {
      this.ws.close();
      this.ws = null;
    }
  }

  private send(event: string, data: object): void {
    if (this.ws && this.ws.readyState === webSocket.WebSocketState.OPEN) {
      this.ws.send(JSON.stringify({ event, data }));
    }
  }

  private handleMessage(event: string, data: object): void {
    const handlers = this.messageHandlers.get(event);
    if (handlers) {
      handlers.forEach(h => h(data as ChatMessage));
    }
  }

  private notifyStatus(status: string): void {
    this.statusHandlers.forEach(h => h(status));
  }

  private tryReconnect(): void {
    if (this.reconnectAttempts < this.maxReconnect) {
      this.reconnectAttempts++;
      const delay = Math.min(1000 * Math.pow(2, this.reconnectAttempts), 30000);
      setTimeout(() => {
        console.info(`WebSocket 重连 (${this.reconnectAttempts}/${this.maxReconnect})`);
        this.connect();
      }, delay);
    }
  }
}

// 全局单例
export const wsService = new WebSocketService();
```

```typescript
function formatTime(isoStr: string): string {
  const date = new Date(isoStr);
  const now = new Date();
  const diffMs = now.getTime() - date.getTime();
  const diffDays = Math.floor(diffMs / (1000 * 60 * 60 * 24));

  if (diffDays === 0) {
    return `${date.getHours().toString().padStart(2, '0')}:${date.getMinutes().toString().padStart(2, '0')}`;
  } else if (diffDays === 1) {
    return '昨天';
  } else if (diffDays < 7) {
    return `${diffDays}天前`;
  } else {
    return `${date.getMonth() + 1}/${date.getDate()}`;
  }
}
```

```typescript
// entry/src/main/ets/components/LocationPicker.ets
import { geoLocationManager } from '@kit.LocationKit';
import { abilityAccessCtrl } from '@kit.AbilityKit';

@Entry
@Component
struct LocationPickerPage {
  @State currentLat: number = 0;
  @State currentLng: number = 0;
  @State searchText: string = '';
  @State suggestions: LocationSuggestion[] = [];
  @State selectedLocation: LocationSuggestion | null = null;

  aboutToAppear() {
    this.getCurrentLocation();
  }

  /**
   * 获取当前位置
   */
  async getCurrentLocation() {
    try {
      // 检查权限
      const atManager = abilityAccessCtrl.createAtManager();
      // ... 权限检查逻辑

      const location = await geoLocationManager.getCurrentLocation({
        priority: geoLocationManager.LocationRequestPriority.FIRST_FIX,
        scenario: geoLocationManager.LocationRequestScenario.NAVIGATION,
        maxAccuracy: 100
      });

      this.currentLat = location.latitude;
      this.currentLng = location.longitude;
    } catch (e) {
      console.error('获取位置失败:', (e as Error).message);
    }

  /**
   * 搜索地点

    if (!this.searchText.trim()) return;

      // 调用后端地点搜索 API
      const res = await httpGet<{ suggestions: LocationSuggestion[] }>(
        '/locations/suggest',

          keyword: this.searchText,
          lng: this.currentLng.toString(),
          lat: this.currentLat.toString()
        }
      );
      this.suggestions = res.suggestions;
    } catch (e) {
      console.error('搜索地点失败');
    }
  }

  /**
   * 选择地点
   */

    this.selectedLocation = loc;
    // 回调给上一页
    router.back();
  }

  build() {
    Column() {
      // 搜索栏

        TextInput({ placeholder: '搜索地点...', text: this.searchText })
          .layoutWeight(1)
          .height(44)
          .onChange((value: string) => { this.searchText = value; })
          .onSubmit(() => this.searchLocations())

        Button('搜索')
          .height(44)
          .onClick(() => this.searchLocations())
      }
      .width('100%')
      .padding(16)

      // 地点列表
      List() {
        ForEach(this.suggestions, (loc: LocationSuggestion) => {
          ListItem() {
            Row() {
              Column() {
                Text(loc.name)
                  .fontSize(16)
                  .fontWeight(FontWeight.Medium)
                Text(`${loc.district} ${loc.address}`)
                  .fontSize(13)
                  .fontColor('#999')
                  .margin({ top: 4 })
              }
              .alignItems(HorizontalAlign.Start)
              .layoutWeight(1)
            }
            .width('100%')
            .padding(16)
          }
          .onClick(() => this.selectLocation(loc))
        })
      }
      .width('100%')
      .layoutWeight(1)
    }
    .width('100%')
    .height('100%')
  }
}

  district: string;
```

**方案 B：自建图形验证码**

```typescript
// entry/src/main/ets/components/Captcha.ets
@Component
export struct CaptchaComponent {
  @Prop captchaId: string;
  @State captchaImage: string = '';  // Base64
  @State userInput: string = '';
  @State verified: boolean = false;

  aboutToAppear() {
    this.loadCaptcha();
  }

  async loadCaptcha() {
    // 从后端获取验证码图片
    const res = await httpGet<{ image: string; token: string }>(
      `/captcha/generate?id=${this.captchaId}`
    );
    this.captchaImage = res.image;
  }

  async verify() {
    try {
      await httpPost('/captcha/verify', {
        token: this.captchaToken,
        code: this.userInput
      });
      this.verified = true;
    } catch {
      this.loadCaptcha(); // 刷新验证码
    }
  }

  build() {
    Row() {
      Image(this.captchaImage)
        .width(120)
        .height(40)
        .onClick(() => this.loadCaptcha())

      TextInput({ placeholder: '验证码', text: this.userInput })
        .width(100)
        .height(40)
        .onChange((v: string) => { this.userInput = v; })

      Button('验证')
        .height(40)
        .onClick(() => this.verify())
    }
  }
}
```

## 八、关键注意事项

### 8.1 网络安全

```
⚠️ 鸿蒙 NEXT 要求所有网络请求使用 HTTPS（明文流量默认禁止）

解决方案：
1. 生产环境：确保后端 API 使用 HTTPS
2. 开发环境：在 module.json5 中配置明文流量例外
   {
     "deviceConfig": {
       "default": {
         "network": {
           "cleartextTraffic": true  // 仅开发环境
         }
       }
     }
   }
```

### 8.2 权限管理

```
⚠️ 鸿蒙 NEXT 的权限分为 system_grant（安装时授权）和 user_grant（运行时申请）

system_grant 权限（自动授权）：
- ohos.permission.INTERNET

user_grant 权限（需运行时申请）：
- ohos.permission.CAMERA
- ohos.permission.READ_IMAGEVIDEO
- ohos.permission.LOCATION
- ohos.permission.APPROXIMATELY_LOCATION
```

运行时权限申请示例：

```typescript
import { abilityAccessCtrl, common } from '@kit.AbilityKit';

async function requestPermission(permission: string): Promise<boolean> {
  try {
    const atManager = abilityAccessCtrl.createAtManager();
    const context = getContext() as common.UIAbilityContext;
    const result = await atManager.requestPermissionsFromUser(context, [permission]);
    return result.authResults[0] === 0;
  } catch {
    return false;
  }
}
```

### 8.3 数据类型差异

```
⚠️ ArkTS 相比 TypeScript 有一些限制

1. 不支持 any 类型（需使用 unknown 或具体类型）
2. 不支持 dynamic import
3. 不支持 eval()
4. 不支持 prototype 修改
5. 推荐使用 struct 而非 class 定义组件
6. 不支持 ?.（可选链）在某些场景下需改为 if 判断
```

### 8.4 包体积优化

```
⚠️ 鸿蒙应用有包体积限制

1. 图片资源使用 WebP 格式（比 PNG 小 30-50%）
2. 大图使用网络加载，不打包到应用中
3. 使用资源限定词按设备加载不同分辨率资源
4. 避免引入不必要的三方库
```

### 8.5 后端 API 兼容性

```
⚠️ 如果后端已迁移到鸿蒙服务化架构

1. API 地址可能变化（从 IP:Port 变为服务域名）
2. 认证方式可能从 JWT 变为华为账号 + 自定义 token
3. 推送通知可能需要接入华为 Push Kit
4. 文件上传可能需要对接华为 OBS 或保持原有方案
```
